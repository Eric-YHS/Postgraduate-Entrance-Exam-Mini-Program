7833	function gradeEntrancePaper(paper, answers) {
7834	  const questions = flattenPaperQuestions(paper);
7835	  let score = 0;
7836	  let objectiveScore = 0;
7837	  let subjectiveScore = null;
7838	  const total = questions.reduce((sum, question) => sum + (Number(question.score) || 2), 0);
7839	  const subjectScores = {};
7840	  const subjectTotals = {};
7841	  const wrongQuestions = [];
7842	  let hasSubjective = false;
7843	
7844	  questions.forEach(question => {
7845	    const points = Number(question.score) || 2;
7846	    const subject = question.subject || paper.subject || '综合';
7847	    subjectTotals[subject] = (subjectTotals[subject] || 0) + points;
7848	    const questionType = getQuestionType(question);
7849	    const objective = ['single_choice', 'multiple_choice', 'true_false', 'fill_blank'].includes(questionType);
7850	    const studentAnswer = answers?.[question.id];
7851	    const correct = objective && answersMatch(studentAnswer, question.correctAnswer ?? question.answer, questionType === 'multiple_choice');
7852	    if (correct) {
7853	      score += points;
7854	      objectiveScore += points;
7855	      subjectScores[subject] = (subjectScores[subject] || 0) + points;
7856	    } else {
7857	      if (!objective) hasSubjective = true;
7858	      wrongQuestions.push({
7859	        questionId: question.id,
7860	        subject,
7861	        number: question.number,
7862	        kind: questionType,
7863	        prompt: question.prompt,
7864	        correctAnswer: question.correctAnswer ?? question.answer,
7865	        studentAnswer: studentAnswer ?? '',
7866	        gradingStatus: objective ? '客观题已判错' : '待批改',
7867	        correctMethod: deriveCorrectMethod(question),
7868	        knowledgePointExplanation: question.knowledge
7869	      });
7870	    }
7871	  });
7872	
7873	  const gradingStatus = hasSubjective ? '部分待批改' : '已批改';
7874	  const aiFeedback = buildEntranceFeedback(subjectScores, subjectTotals, objectiveScore, total);
7875	  return { score: hasSubjective ? null : score, objectiveScore, subjectiveScore, total, gradingStatus, subjectScores, subjectTotals, wrongQuestions, aiFeedback };
7876	}
