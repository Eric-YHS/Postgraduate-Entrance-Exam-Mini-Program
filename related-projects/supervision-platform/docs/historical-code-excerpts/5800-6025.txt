5893	const MathFormula = ({value}) => {
5894	  const source = String(value || '').trim();
5895	  if (!source) return <span className="math-formula-empty">—</span>;
5896	  const superscriptMap = {'⁰':'0','¹':'1','²':'2','³':'3','⁴':'4','⁵':'5','⁶':'6','⁷':'7','⁸':'8','⁹':'9','⁺':'+','⁻':'-','⁽':'(','⁾':')'};
5897	  const subscriptMap = {'₀':'0','₁':'1','₂':'2','₃':'3','₄':'4','₅':'5','₆':'6','₇':'7','₈':'8','₉':'9','₊':'+','₋':'-','₍':'(','₎':')','ₓ':'x'};
5898	  const normalize = text => {
5899	    let result = text.replace(/^[\$]+|[\$]+$/g, '').trim()
5900	      .replace(/−/g, '-').replace(/×/g, '\\times ').replace(/÷/g, '\\div ')
5901	      .replace(/→/g, '\\to ').replace(/≤/g, '\\le ').replace(/≥/g, '\\ge ').replace(/∞/g, '\\infty ')
5902	      .replace(/√/g, '\\sqrt{}');
5903	    result = result.replace(/[⁰¹²³⁴⁵⁶⁷⁸⁹⁺⁻⁽⁾]+/g, match => `^{${[...match].map(char => superscriptMap[char]).join('')}}`);
5904	    result = result.replace(/[₀₁₂₃₄₅₆₇₈₉₊₋₍₎ₓ]+/g, match => `_{${[...match].map(char => subscriptMap[char]).join('')}}`);
5905	    return result;
5906	  };
5907	  const normalized = normalize(source);
5908	  if (/[㐀-鿿]/.test(normalized) && !/[\\_^{}]/.test(normalized)) return <span className="math-formula-text">{source}</span>;
5909	  try {
5910	    return <span className="math-formula" dangerouslySetInnerHTML={{__html:katex.renderToString(normalized, {throwOnError:false, displayMode:false, trust:false, strict:'warn'})}}/>;
5911	  } catch (error) {
5912	    return <span className="math-formula-fallback">{source}</span>;
5913	  }
