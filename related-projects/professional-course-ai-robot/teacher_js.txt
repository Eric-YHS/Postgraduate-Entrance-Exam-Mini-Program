async function fetchTeacherData(url, options = {}) {
  const controller = new AbortController();
  const timeout = setTimeout(() => controller.abort(), options.timeoutMs || 5000);
  try {
    const { timeoutMs, ...requestOptions } = options;
    return await fetch(url, { ...requestOptions, signal: controller.signal });
  } finally {
    clearTimeout(timeout);
  }
}

async function loadTeacherWorkspace() {
  const base = teacherApiBase;
  adminBase = base;
  adminToken = '';
  const initialStatus = $('#student-sync-status');
  if (initialStatus) initialStatus.textContent = '正在连接本地教学服务…';
  let requests;
  try {
    const health = await fetchTeacherData(`${base}/health`, { headers: authHeaders(), cache: 'no-store', timeoutMs: 2500 });
    if (!health.ok) throw new Error(`教学服务返回 HTTP ${health.status}`);
    requests = await Promise.allSettled([
      fetchTeacherData(`${base}/admin/overview`, { headers: authHeaders(), cache: 'no-store' }),
      fetchTeacherData(`${base}/admin/students`, { headers: authHeaders(), cache: 'no-store' }),
      fetchTeacherData(`${base}/admin/knowledge-base`, { headers: authHeaders(), cache: 'no-store' }),
      fetchTeacherData(`${base}/admin/settings`, { headers: authHeaders(), cache: 'no-store' }),
    ]);
  } catch (error) {
    console.error('Teacher API is unavailable:', error);
    renderStudentReadError('教学服务未启动。请启动 server.py 后重新读取。');
    $('#roles').innerHTML = row('教学服务', false, '未启动');
    $('#caps').innerHTML = '';
    return;
  }
  const responseAt = index => requests[index]?.status === 'fulfilled' && requests[index].value.ok ? requests[index].value : null;
  const [overviewResponse, studentsResponse, knowledgeResponse, settingsResponse] = [0, 1, 2, 3].map(responseAt);
  let studentLoaded = false;
  try {
    if (studentsResponse) {
      const studentData = await studentsResponse.json();
      renderStudents(studentData.students || []);
      studentLoaded = true;
    }
  } catch (error) { console.error('Student roster refresh failed:', error); }
  if (!studentLoaded) {
    renderStudentReadError('学生档案暂时无法读取，请重新读取。');
  }
  try {
    const [data, knowledgeData, settings] = await Promise.all([
      overviewResponse ? overviewResponse.json() : Promise.resolve({}),
      knowledgeResponse ? knowledgeResponse.json() : Promise.resolve({}),
      settingsResponse ? settingsResponse.json() : Promise.resolve({}),
    ]);
    const docs = data.documents || {}, service = data.service || {}, students = data.students || {};

    $('#knowledge-count').textContent = `${data.knowledgeBase?.entries || 0} 个归档`;
    $('#roles').innerHTML = roles.join('') || row('模型服务', false);
    $('#caps').innerHTML = [row('联网搜索', service.webSearchConfigured), row('OCR', service.ocrConfigured)].join('');
    const capabilityNotes = service.capabilityNotes || {};
    $('#operations-roles').innerHTML = [
      row(`文本模型（${capabilityNotes.text || '学习工具调用'}）`, service.modelRoles?.text?.configured, service.modelRoles?.text?.displayName || service.modelRoles?.text?.model || ''),
      row(`视觉模型（${capabilityNotes.vision || '图片识别'}）`, service.modelRoles?.vision?.configured, service.modelRoles?.vision?.displayName || service.modelRoles?.vision?.model || ''),

    $('#roles').innerHTML = row('教学数据', false, '暂时无法读取');
    $('#caps').innerHTML = '';
    $('#parse-status').textContent = '待更新';
    $('#knowledge-count').textContent = '待更新';
  }

/* Model-library controls: connection deletion and unambiguous verification output. */
function installModelLibraryDeletion() {
  const panel = $('#model-library-editor');
  const form = $('#library-model-form');
  if (!panel || !form || panel.hidden || form.dataset.deleteReady === '1') return;
  form.dataset.deleteReady = '1';
  const id = String(form.elements.id?.value || '').trim();
  if (!id) return;
  const actions = form.querySelector('.model-editor-actions');
  if (!actions) return;
  const remove = document.createElement('button');
  remove.type = 'button'; remove.className = 'secondary'; remove.id = 'delete-library-model'; remove.textContent = '删除此模型';
  actions.insertBefore(remove, actions.firstChild);
  remove.onclick = async () => {
    const name = form.elements.connectionName?.value.trim() || '该模型连接';
    if (!confirm(`确定删除“${name}”吗？\n\n会解除使用它的默认路由和机器人绑定；学生、资料和学习记录不会受影响。`)) return;
    remove.disabled = true;
    const note = $('#library-model-note'); if (note) note.textContent = '正在删除…';
    try {
      const response = await fetch(`${adminBase}/admin/models/${encodeURIComponent(id)}`, { method: 'DELETE', headers: authHeaders() });
      const result = await response.json();
      if (!response.ok) throw new Error(result.message || '删除失败');
      panel.hidden = true;
      fillSettings(result.settings || await fetch(`${adminBase}/admin/settings`, { headers: authHeaders() }).then(item => item.json()));
    } catch (error) {
      remove.disabled = false;
      if (note) note.textContent = `删除失败：${error.message}`;
    }
  };
  const verify = $('#verify-library-model');
  if (verify) verify.addEventListener('click', () => {
    const note = $('#library-model-note');
    if (note) note.textContent = `正在验证“${form.elements.connectionName?.value.trim() || '当前连接'}”及模型 ${form.elements.upstreamModel?.value || '—'}…`;
  }, true);
}
const modelLibraryObserver = new MutationObserver(installModelLibraryDeletion);
modelLibraryObserver.observe(document.body, { childList: true, subtree: true });
installModelLibraryDeletion();
